cmake_minimum_required(VERSION 3.20)
project(CPNetFluxDesktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(cpnetflux-desktop ui/main.cpp ui/rpc_client.cpp ui/rpc_client.h)
qt_add_resources(cpnetflux-desktop qml PREFIX "/" FILES ui/Main.qml)
target_link_libraries(cpnetflux-desktop PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/agent" "${CMAKE_BINARY_DIR}/schemas")
configure_file(agent/cpnetflux_agent.py "${CMAKE_BINARY_DIR}/agent/cpnetflux_agent.py" COPYONLY)
configure_file(schemas/desktop-agent.schema.json "${CMAKE_BINARY_DIR}/schemas/desktop-agent.schema.json" COPYONLY)
file(WRITE "${CMAKE_BINARY_DIR}/cpnetflux-agent" "#!/bin/sh\nexec '${Python3_EXECUTABLE}' '${CMAKE_BINARY_DIR}/agent/cpnetflux_agent.py' \"$@\"\n")
file(CHMOD "${CMAKE_BINARY_DIR}/cpnetflux-agent" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
add_custom_target(cpnetflux-agent ALL DEPENDS "${CMAKE_BINARY_DIR}/agent/cpnetflux_agent.py")
enable_testing()
add_test(NAME desktop-contract COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_contract.py)
add_test(NAME desktop-agent COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_agent.py)
add_test(NAME desktop-offscreen COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ui.py $<TARGET_FILE:cpnetflux-desktop> ${CMAKE_CURRENT_SOURCE_DIR}/agent/cpnetflux_agent.py)
set_tests_properties(desktop-agent desktop-offscreen PROPERTIES TIMEOUT 40)
